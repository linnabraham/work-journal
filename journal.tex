  \IfStrEq{\todoIU}{linu}{%
	\stepcounter{distractioncount}

	\addcontentsline{tdo}{section}{\textcolor{magenta}{(Today)}~\thetodonotesnumber:\textcolor{\todocolor}{~ #2}}%

}
  \fi
\fi

  \ifTodoIsBacklog

\fi

  }\par
}

\newcommandx{\highlighttodo}[3][1=blue,2=]{%
  \HighlightTodoIsDonefalse % reset
  \setkeys{highlighttodo}{#2}%

    \phantomsection
    \ifHighlightTodoIsDone
    \par\noindent
    \fcolorbox{#1!50!black}{#1!5}{%
      \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule}
      \textbf{Todo \ref{#3}}:\par
	  \sout{\textit{\csname todocontent@#3\endcsname}}
      \end{minipage}
    }\par

	\else

      \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule}

      \end{minipage}
    }\par
    \fi
  }%
