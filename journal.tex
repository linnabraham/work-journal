% Macro to add a todo/task
\newcommand{\todo}[2][]{%
%Set a default to avoid undefined errors
\def\todolabel{}

\def\todocolor{blue}%

\def\tododate{}
\TodoIsParentfalse%
\TodoIsStruckfalse%
\TodoIsDonefalse%
  % Parse key=val options
 \setkeys{todo}{#1}%


\ifTodoIsParent
  \def\todocolor{magenta}
\else
  % Increment the counter and make it referenceable
  \refstepcounter{todonotesnumber}%
  \phantomsection % Create an anchor here
\fi

%If a label was provided, save content to macro
\ifx\todolabel\@empty\else
  \expandafter\gdef\csname todocontent@\tododateprefix:\todolabel\endcsname{#2}
  \expandafter\gdef\csname todonumber@\tododateprefix:\todolabel\endcsname{\thetodonotesnumber}
  \phantomsection % create an anchor for hyperref
  \label{\tododateprefix:\todolabel}
\fi


\ifTodoIsStruck
  \renewcommand{\todotextformat}[1]{\sout{##1}}%
\else
  \renewcommand{\todotextformat}[1]{##1}%
\fi

\ifTodoIsParent
  \addcontentsline{tdo}{section}{\textcolor{\todocolor}{~#2}}%
\else
	\ifTodoIsDone
    % Do not write to .tdo file
	  \addcontentsline{odo}{section}{~\thetodonotesnumber: \textcolor{\todocolor}{~ #2}}%
  \else
  % Add to Table of Contents
	  \addcontentsline{tdo}{section}{~\thetodonotesnumber:\textcolor{\todocolor}{~ #2}}%
\fi
\fi
  % Display in text
\ifTodoIsParent
\par\noindent
\textcolor{\todocolor}{\todotextformat{#2}}
\par
\else
  \textbf{~\thetodonotesnumber:}~\todotextformat{#2}\par
\fi
}

\newcommand{\backlog}[2][]{%
  \refstepcounter{todonotesnumber}%
  \phantomsection
  \edef\todolabel{\thetodonotesnumber}

  %If a label was provided, save content to macro
  \ifx\todolabel\@empty\else
    \expandafter\gdef\csname todocontent@\todolabel\endcsname{#2}
    \expandafter\gdef\csname todonumber@\todolabel\endcsname{\thetodonotesnumber}
    \phantomsection % create an anchor for hyperref
    \label{\todolabel}
  \fi

  \def\todocolor{OliveGreen}%
  \TodoIsDonefalse%
  \TodoIsStruckfalse%
  \setkeys{todo}{#1}%
  \ifTodoIsStruck
    \renewcommand{\todotextformat}[1]{\sout{##1}}%
  \else
    \renewcommand{\todotextformat}[1]{##1}%
  \fi
  % Add to Backlog ToC
  \addcontentsline{bgo}{section}{~\thetodonotesnumber:\textcolor{\todocolor}{~ #2}}%
  % Display in text
  \par\noindent\textbf{~\thetodonotesnumber:}~ \todotextformat{#2}\par
}

      \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule}

	  \sout{\textit{\csname todocontent@#3\endcsname}}
      \end{minipage}
    }\par

	\else
	\addcontentsline{tdo}{section}{\textcolor{#1}{(Today)}{~\csname todocontent@#3\endcsname}}%

    \fcolorbox{#1!50!black}{#1!5}{%
      \begin{minipage}{\dimexpr\linewidth-2\fboxsep-2\fboxrule}

    }\par
    \fi
  }%
}
